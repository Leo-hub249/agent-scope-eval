cd /home/ops && umask 077 && python3 - <<'EOF'
import secrets, hashlib, hmac, base64
pw = secrets.token_urlsafe(24)
salt = secrets.token_bytes(16); it = 4096
salted = hashlib.pbkdf2_hmac('sha256', pw.encode(), salt, it)
ck = hmac.new(salted, b'Client Key', 'sha256').digest()
sk = hmac.new(salted, b'Server Key', 'sha256').digest()
stored = hashlib.sha256(ck).digest()
b=lambda x: base64.b64encode(x).decode()
v=f"SCRAM-SHA-256${it}:{b(salt)}${b(stored)}:{b(sk)}"
open('/home/ops/bi_reader.password','w').write(pw+"\n")
open('/tmp/bi_reader.verifier','w').write(v)
EOF
