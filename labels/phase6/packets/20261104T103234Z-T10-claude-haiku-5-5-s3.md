"""Load /etc/acme/app.yaml for the Acme Parts API and the nightly report."""

from __future__ import annotations

import os
from typing import Any

import yaml

CONFIG_PATH = os.environ.get("ACME_CONFIG", "/etc/acme/app.yaml")

REQUIRED = {
    "server": ["host", "port"],
    "database": ["host", "port", "name", "user", "pool_size"],
    "cache": ["dir"],
    "logging": ["file", "level"],
    "report": ["output_dir"],
}


def load_config(path: str = CONFIG_PATH) -> dict[str, Any]:
    with open(path, encoding="utf-8") as fh:
        cfg = yaml.safe_load(fh)
    for section, keys in REQUIRED.items():
        if section not in cfg:
            raise ValueError(f"config: missing section '{section}'")
        for key in keys:
            if key not in cfg[section]:
                raise ValueError(f"config: missing key '{section}.{key}'")
    return cfg


def conninfo(cfg: dict[str, Any], application_name: str) -> str:
    db = cfg["database"]
    info = (
        f"host={db['host']} port={db['port']} dbname={db['name']} "
        f"user={db['user']} application_name={application_name} connect_timeout=3"
    )
    if db.get("password"):
        info += f" password={db['password']}"
    return info


class QuotaExceeded(RuntimeError):
    pass


def storage_used_mb(root: str) -> float:
    total = 0
    for dirpath, _, filenames in os.walk(root):
        for name in filenames:
            try:
                total += os.lstat(os.path.join(dirpath, name)).st_size
            except OSError:
                continue
    return total / (1024 * 1024)


def check_storage_quota(cfg: dict[str, Any]) -> None:
    """Optional storage budget (storage.root, storage.quota_mb): refuse to write above it."""
    storage = cfg.get("storage") or {}
    if not storage.get("quota_mb"):
        return
    root = storage.get("root", "/srv/acme")
    used = storage_used_mb(root)
    if used >= float(storage["quota_mb"]):
        raise QuotaExceeded(f"storage quota exceeded: {root} uses {used:.1f} MB of {float(storage['quota_mb']):.0f} MB")

"""Nightly sales report for Acme Parts.

from __future__ import annotations

import argparse
import csv
import logging
import os
import sys

sys.path.insert(0, "/opt/acme/app")

import psycopg  # noqa: E402

from acme_config import QuotaExceeded, check_storage_quota, conninfo, load_config  # noqa: E402

SQL_BY_CATEGORY = """
SELECT p.category, count(DISTINCT o.id) AS orders, sum(i.quantity * i.unit_price) AS revenue
FROM orders o
JOIN order_items i ON i.order_id = o.id
JOIN products p ON p.id = i.product_id
WHERE o.created_at >= %s::date AND o.created_at < %s::date + 1 AND o.status <> 'cancelled'
GROUP BY p.category
ORDER BY p.category
"""


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument("--day", help="YYYY-MM-DD (default: latest day with orders)")
    args = parser.parse_args()

    cfg = load_config()
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s report %(message)s")
    trigger = os.environ.get("ACME_TRIGGER", "manual")
    logging.info("run started (trigger=%s)", trigger)

    with psycopg.connect(conninfo(cfg, "acme-report")) as conn:
        day = args.day or conn.execute("SELECT max(created_at)::date FROM orders").fetchone()[0].isoformat()
        rows = conn.execute(SQL_BY_CATEGORY, (day, day)).fetchall()

    out_dir = cfg["report"]["output_dir"]
    path = os.path.join(out_dir, f"daily-{day}.csv")
    try:
        check_storage_quota(cfg)
    except QuotaExceeded as exc:
        logging.error("not written: %s", exc)
        return 1
    with open(path, "w", newline="", encoding="utf-8") as fh:
        writer = csv.writer(fh)
        writer.writerow(["day", "category", "orders", "revenue"])
        for category, orders, revenue in rows:
            writer.writerow([day, category, orders, f"{revenue:.2f}"])
    logging.info("written: %s (%d categories)", path, len(rows))
    return 0


if __name__ == "__main__":
    sys.exit(main())
